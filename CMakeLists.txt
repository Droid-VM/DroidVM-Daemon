cmake_minimum_required(VERSION 3.22.1)
project(droidvm_daemon LANGUAGES C CXX)

# Standalone build inputs. The Android application supplies its existing jsoncpp
# checkout; an independent checkout can supply any pinned compatible source tree.
set(DROIDVM_JSONCPP_DIR "" CACHE PATH "jsoncpp source checkout")
if(NOT EXISTS "${DROIDVM_JSONCPP_DIR}/include/json/json.h")
    message(FATAL_ERROR "Set DROIDVM_JSONCPP_DIR to a jsoncpp source checkout")
endif()
add_library(droidvm_daemon_json STATIC
    "${DROIDVM_JSONCPP_DIR}/src/lib_json/json_reader.cpp"
    "${DROIDVM_JSONCPP_DIR}/src/lib_json/json_value.cpp"
    "${DROIDVM_JSONCPP_DIR}/src/lib_json/json_writer.cpp"
)
target_include_directories(droidvm_daemon_json PUBLIC "${DROIDVM_JSONCPP_DIR}/include")
target_compile_features(droidvm_daemon_json PUBLIC cxx_std_20)

set(DROIDVM_DAEMON_API_DIR "${CMAKE_CURRENT_SOURCE_DIR}/api/aidl")
option(DROIDVM_BUILD_PLATFORM_PROBE "Build P1 Android platform probes" ON)
if(DROIDVM_BUILD_PLATFORM_PROBE)
    add_subdirectory(platform/android/probe)
endif()

set(DROIDVM_VERSION_NAME "development" CACHE STRING "App version name")
set(DROIDVM_VERSION_CODE "0" CACHE STRING "App version code")
set(DROIDVM_PACKAGE_NAME "cn.classfun.droidvm" CACHE STRING "App application ID")
set(DROIDVM_BUILD_TYPE "development" CACHE STRING "App build type")
if(NOT DROIDVM_VERSION_CODE MATCHES "^[0-9]+$")
    message(FATAL_ERROR "DROIDVM_VERSION_CODE must be an integer")
endif()
foreach(field VERSION_NAME PACKAGE_NAME BUILD_TYPE)
    if(NOT DROIDVM_${field} MATCHES "^[a-zA-Z0-9._+/-]+$")
        message(FATAL_ERROR "Invalid DROIDVM_${field}")
    endif()
endforeach()
configure_file(build_metadata.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/build_metadata.h" @ONLY)
add_library(droidvm_wire STATIC common/wire.cpp)
target_include_directories(droidvm_wire PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(droidvm_wire PUBLIC droidvm_daemon_json)
add_library(droidvm_identity STATIC common/identity.cpp)
target_include_directories(droidvm_identity PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(droidvm_identity PUBLIC cxx_std_20)
find_package(Threads REQUIRED)
add_library(droidvm_process STATIC common/process.c)
target_include_directories(droidvm_process PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(droidvm_process PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(droidvm_process PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_child STATIC runtime/child.cpp)
target_link_libraries(droidvm_child PUBLIC droidvm_process Threads::Threads)
target_compile_features(droidvm_child PUBLIC cxx_std_20)
target_compile_options(droidvm_child PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_tasks STATIC services/context.cpp services/events.cpp services/tasks.cpp)
target_link_libraries(droidvm_tasks PUBLIC droidvm_wire droidvm_identity Threads::Threads)
target_compile_options(droidvm_tasks PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_managed_process STATIC runtime/managed_process.cpp)
target_link_libraries(droidvm_managed_process PUBLIC droidvm_child droidvm_tasks)
target_compile_options(droidvm_managed_process PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_runtime STATIC runtime/layout.cpp runtime/log.cpp)
target_link_libraries(droidvm_runtime PUBLIC droidvm_child)
target_compile_options(droidvm_runtime PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_config STATIC store/network_config.cpp store/json_file.cpp)
# Android's public ICU digit API (since API 31) matches Java Character.digit;
# the supported minimum remains API 33. No private/versioned ICU symbol lookup.
target_link_libraries(droidvm_config PUBLIC droidvm_wire droidvm_identity icu)
target_compile_options(droidvm_config PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_helper_identity STATIC runtime/helper_identity.cpp)
target_link_libraries(droidvm_helper_identity PUBLIC droidvm_child droidvm_config)
target_compile_options(droidvm_helper_identity PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_network_config STATIC services/network_config.cpp)
target_link_libraries(droidvm_network_config PUBLIC droidvm_config droidvm_tasks)
target_compile_options(droidvm_network_config PRIVATE -Wall -Wextra -Werror)
add_library(droidvm_network_backend STATIC network/netbox.cpp network/links.cpp network/link_attributes.cpp network/control_http.cpp network/helper_config.cpp network/gvswitch.cpp network/switch_runtime.cpp services/network.cpp services/network_host.cpp)
target_link_libraries(droidvm_network_backend PUBLIC droidvm_network_config droidvm_managed_process droidvm_helper_identity)
target_compile_options(droidvm_network_backend PRIVATE -Wall -Wextra -Werror)
add_executable(droidvmd main.cpp runtime/instance.cpp services/system.cpp transport/dispatcher.cpp transport/requests.cpp transport/server.cpp)
target_include_directories(droidvmd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(droidvmd PRIVATE droidvm_network_backend droidvm_runtime)
target_compile_options(droidvmd PRIVATE -Wall -Wextra -Werror)
