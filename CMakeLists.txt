# HostSocName's chip table: the app's res/xml/qcom.xml stays the only source.
set(DROIDVM_QCOM_XML "${DROIDVM_JSONCPP_DIR}/../../res/xml/qcom.xml" CACHE FILEPATH "App chip-name table")
if(NOT EXISTS "${DROIDVM_QCOM_XML}")
    message(FATAL_ERROR "Set DROIDVM_QCOM_XML to the app's res/xml/qcom.xml")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DROIDVM_QCOM_XML}")
file(READ "${DROIDVM_QCOM_XML}" DROIDVM_QCOM_TEXT)
string(REGEX REPLACE "<!--([^-]|-[^-])*-->" "" DROIDVM_QCOM_TEXT "${DROIDVM_QCOM_TEXT}")
string(REGEX MATCHALL "<chip id=\"[^\"]*\"[^>]*>[^<]*</chip>" DROIDVM_QCOM_CHIPS "${DROIDVM_QCOM_TEXT}")
set(DROIDVM_QCOM_ROWS "")
foreach(chip IN LISTS DROIDVM_QCOM_CHIPS)
    string(REGEX REPLACE "<chip id=\"([^\"]*)\"[^>]*>([^<]*)</chip>" "\\1" chip_id "${chip}")
    string(REGEX REPLACE "<chip id=\"([^\"]*)\"[^>]*>([^<]*)</chip>" "\\2" chip_name "${chip}")
    string(STRIP "${chip_id}" chip_id)
    string(STRIP "${chip_name}" chip_name)
    string(REPLACE "\\" "\\\\" chip_name "${chip_name}")
    string(REPLACE "\"" "\\\"" chip_name "${chip_name}")
    string(APPEND DROIDVM_QCOM_ROWS "    {\"${chip_id}\", \"${chip_name}\"},\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/qcom_chips.h.tmp"
    "#pragma once\n#include <utility>\n// Generated from ${DROIDVM_QCOM_XML}; later rows win.\ninline constexpr std::pair<const char*, const char*> qcom_chips[] = {\n${DROIDVM_QCOM_ROWS}};\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/generated/qcom_chips.h.tmp" "${CMAKE_CURRENT_BINARY_DIR}/generated/qcom_chips.h" COPYONLY)
